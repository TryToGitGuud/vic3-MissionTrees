mt_fra_placate_legitimists = {
	is_shown = {
		c:FRA ?= THIS
		has_variable = mt_fra_unlock_placate_legitimists_decision
		NOT = { has_variable = mt_fra_placate_legitimists_taken }
	}
	possible = { any_political_movement = { is_political_movement_type = movement_legitimist } }
	when_taken = {
		set_variable = mt_fra_placate_legitimists_taken
		remove_variable = mt_fra_unlock_placate_legitimists_decision
		random_political_movement = {
			limit = { is_political_movement_type = movement_legitimist }
			add_modifier = { name = mt_fra_legitimist_placated days = long_modifier_time }
		}
		trigger_event = { id = mt_french_missions.6 days = 0 }
	}
	ai_chance = { value = 0 }
}
