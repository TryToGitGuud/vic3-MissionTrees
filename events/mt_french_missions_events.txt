mt_french_missions.6 = {
	type = country_event
	placement = ROOT
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = mt_french_missions.6.t
	desc = mt_french_missions.6.d
	flavor = mt_french_missions.6.f

	option = {
		name = mt_french_missions.6.a
		default_option = yes
		custom_tooltip = legitimist_progress_increase_40_tt
		je:je_divided_monarchists ?= { add_progress = { name = divided_monarchists_legitimist_progress_bar value = 40 } }
		ig:ig_devout ?= { add_modifier = { name = mt_fra_devout_legitimist_favor days = long_modifier_time } }
	}
	option = {
		name = mt_french_missions.6.b
		custom_tooltip = legitimist_progress_increase_10_tt
		je:je_divided_monarchists ?= { add_progress = { name = divided_monarchists_legitimist_progress_bar value = 10 } }
	}
	option = {
		name = mt_french_missions.6.c
		custom_tooltip = mt_fra_legitimist_progress_decrease_40_tt
		je:je_divided_monarchists ?= { add_progress = { name = divided_monarchists_legitimist_progress_bar value = -40 } }
		ig:ig_devout ?= { add_modifier = { name = mt_fra_devout_legitimist_displeasure days = long_modifier_time } }
		if = {
			limit = { any_political_movement = { is_political_movement_type = movement_legitimist } }
			random_political_movement = {
				limit = { is_political_movement_type = movement_legitimist }
				add_modifier = { name = mt_fra_legitimist_secular_setback days = long_modifier_time }
			}
		}
	}
}
